{{- /* Emits a or that queries the network in the browser rather than at build time. Content adapters run only at build time, so anything that must be current on every visit lives here instead. This is the same trade the standalone builder called `live` mode. Args: kind "list" (default) or "table" section id of a section in params.nanopub.sections — supplies everything below query query nanopub URI or query template id (when no section is given); overrides the section's own query or view params JSON string of query parameters titleField, dateField, columns, limit, groupByYear */ -}} {{- $np := site.Params.nanopub -}} {{- $kind := .kind | default "list" -}} {{- $query := .query -}} {{- $params := .params -}} {{- $titleField := .titleField -}} {{- $dateField := .dateField -}} {{- $columns := .columns -}} {{- $limit := .limit -}} {{- $groupByYear := .groupByYear -}} {{- $label := "nanopub live element" -}} {{- /* Resolved here rather than below, because a section given by id may name a view instead of a query — and a view also supplies the parameter that carries the subject, which has to be in $params before they are serialised for the browser. */ -}} {{- $qid := "" -}} {{- with .section -}} {{- $sec := partial "nanopub/section-config.html" (dict "id" .) -}} {{- $label = printf "section %q" . -}} {{- $fields := $sec.fields | default dict -}} {{- $titleField = $titleField | default $fields.title -}} {{- $dateField = $dateField | default $fields.date -}} {{- $limit = $limit | default $sec.limit -}} {{- if and (not $columns) $sec.columns -}} {{- $columns = $sec.columns | jsonify -}} {{- end -}} {{- $source := partial "nanopub/section-source.html" (dict "section" $sec "label" $label) -}} {{- if not $query -}}{{- $qid = $source.queryId -}}{{- end -}} {{- if not $params -}} {{- $qp := dict -}} {{- range $k, $v := $source.params -}} {{- $val := replace (printf "%v" $v) "$ORCID" ($np.orcid | default "") -}} {{- $val = replace $val "$PUBKEYS" ($np.pubkeys | default "") -}} {{- $qp = merge $qp (dict $k $val) -}} {{- end -}} {{- with $qp -}}{{- $params = . | jsonify -}}{{- end -}} {{- end -}} {{- end -}} {{- /* An explicit `query` argument overrides whatever the section resolved to. */ -}} {{- with $query -}} {{- $qid = partial "nanopub/resolve-query.html" (dict "query" . "label" $label) -}} {{- end -}} {{- if not $qid -}} {{- errorf "nanopub-hugo: live element needs either a section id or a query" -}} {{- end -}} {{- partial "nanopub/elements-script.html" . -}} {{- if eq $kind "table" -}} {{- else -}} {{- end -}}
